  });

  // ═══ C. late invoices ══════════════════════════════════════════════════════════════════════════════════════════
  // C1: a late ticket's sold rows sit on the ticket day; a round recorded after it with more dead than left must make the invoice refuse live_negative.

  // C2: cash box, statement and reports by day for a late invoice paid in part.
  let late = null;

      [[['invoice', TODAY], ['payment', TODAY]], [TODAY], [day(-9)], true, {}], 'C2: late invoice and its cash on the issue day, sold rows on the ticket day, the ticket day box unchanged, invariants zero');

  // C3: void and re-post a late invoice on a later day, and replays.
  await section('C3 void/repost/replay', async () => {

  // C4 (O7 regression probe): rounds the system expected between the ticket day and the invoice erase the FCR once the invoice posts.
  await section('C4 rounds before invoice vs FCR', async () => {
    const h = await hall('C4');
    const F = await place(h, day(-6), 100);
    await issueOk(F, X, '300', day(-6), W.feed);
    for (const d of [-6, -5, -4]) await at(day(d), () => round(F, day(d), 0, '20'));
    await at(day(-4), () => one(U.manager, 'select public.record_flock_weight($1, $2, $3, $4, $5)', [F, day(-4), 50, '1000.0', uid()]));

  // ═══ D. advisory locks on stop ids ═════════════════════════════════════════════════════════════════════════════
  // D1: two NEW tickets sharing two stop ids in opposite order, both held behind a flock lock: no deadlock, one wins, the other id_conflict.
  await section('D1 opposite order', async () => {
    const h1 = await hall('D1a');
    const h2 = await hall('D1b');
    const F1 = await place(h1, day(-10), 100);
    const F2 = await place(h2, day(-10), 100);
    const s1 = uid();
    const s2 = uid();
    const c = await db.session(U.manager);
    const a = await db.session(U.worker);
    const b = await db.session(U.manager2);
    try {
      const lo = [F1, F2].sort()[0];
      const held = await c.run(`${ADJUST} as v`, [uid(), uid(), lo, TODAY, 1, 'hold', 100]);
      const TA = uid();
      const TB = uid();
      const pa = a.run(`${SAVE} as v`, [TA, null, JSON.stringify(scaleTicket(S, TODAY, [[s1, h1, 10, '7020'], [s2, h2, 10, '7040']]))]);
      const sa = await settleOn(pa, a.pid, c.pid);
      const pb = b.run(`${SAVE} as v`, [TB, null, JSON.stringify(scaleTicket(S, TODAY, [[s2, h2, 10, '7020'], [s1, h1, 10, '7040']]))]);
      const sb = await settleOn(pb, b.pid, a.pid);
      await c.commit();
      const ra = await pa;
      if (ra.ok) await a.commit();
      else await a.rollback();
      const rb = await pb;
      if (rb.ok) await b.commit();
      else await b.rollback();
      const rows = await admin('select ticket_id from public.weigh_ticket_stops where id = any ($1::uuid[])', [[s1, s2]]);
      log('D1', { held: held.ok, sa, sb, ra: short(ra), rb: short(rb), owners: rows.map((r) => r.ticket_id === TA) });
      t.same([held.ok, sa, sb, short(ra), short(rb), rows.map((r) => r.ticket_id === TA)], [true, 'blocked', 'blocked', 'ok', 'id_conflict', [true, true]],
        'D1: opposite stop order: no deadlock, second id_conflict');
      if (ra.ok) await one(U.manager, 'select public.void_weigh_ticket($1, $2, $3)', [TA, null, uid()]);
    } finally {
      await c.rollback();
      await a.rollback();
      await b.rollback();
    }
  });

  // D2: weighing-id buckets are global: a crate-scale save in ANOTHER org waits on this org's save that waits on flock locks, and fails with a raw lock timeout.
  await section('D2 cross-org buckets', async () => {
    const G2 = await makeOrg(db, { name: 'zz v28 org two', id: id2, phone: phone2, roles: [['owner', 'owner'], ['manager', 'manager']] });
    const S2 = (await call(db, G2.users.owner, "insert into public.sites (division_id, name, kind) values ($1, 'zz site two', 'farm') returning id", [G2.divisions.broiler])).value;

    const ha = await hall('D2a');
    const hb = await hall('D2b');
    const Fa = await place(ha, day(-10), 1000);
    const Fb = await place(hb, day(-10), 1000);
    const [lo, hi] = [Fa, Fb].sort();
    const weighings = (n) => Array.from({ length: n }, () => ({ id: uid(), crates: 10, gross_kg: '300' }));
    const crate = (site, stops) => ({ site_id: site, on_date: TODAY, method: 'crates', empty_kg: null, empty_with_crates: null, crate_kg: '6',
      stops: stops.map(([hallId, n]) => ({ id: uid(), hall_id: hallId, birds: 100, crates: null, gross_kg: null, weighings: weighings(n) })) });
    const c1 = await db.session(U.manager);
    const c2 = await db.session(U.manager2);
    const a = await db.session(U.worker);
    const x = await db.session(G2.users.manager);
    try {
      await c1.run(`${ADJUST} as v`, [uid(), uid(), lo, TODAY, 1, 'hold', 1000]);
      await c2.run(`${ADJUST} as v`, [uid(), uid(), hi, TODAY, 1, 'hold', 1000]);
      const pa = a.run(`${SAVE} as v`, [uid(), null, JSON.stringify(crate(S, [[ha, 150], [hb, 150]]))]);
      const sa = await settleOn(pa, a.pid, c1.pid);
      const t0 = Date.now();
      const px = x.run(`${SAVE} as v`, [uid(), null, JSON.stringify(crate(S2, [[hO, 150]]))]);
      const sx = await settleOn(px, x.pid, a.pid);
      setTimeout(() => c1.commit().catch(() => {}), 3500);
      setTimeout(() => c2.commit().catch(() => {}), 7500);
      const rx = await px;
      const waited = Date.now() - t0;
      if (rx.ok) await x.commit();
      else await x.rollback();
      const ra = await pa;
      if (ra.ok) await a.commit();
      else await a.rollback();
      log('D2', { sa, sx, rx: short(rx), waitedMs: waited, ra: short(ra) });
      t.same([sa, sx, short(rx)], ['blocked', 'finished', 'ok'],
        `D2: a crate-scale save in another org must not wait on (or time out behind) this org's save (got ${sx}, ${short(rx)} after ${waited} ms)`);
    } finally {
      await new Promise((resolve) => setTimeout(resolve, 8000));
      for (const s of [c1, c2, a, x]) await s.rollback().catch(() => {});
    }
  });

  // C5 (last: the period lock touches every date): a late ticket's day inside a locked period refuses period_locked; lock below it posts.
