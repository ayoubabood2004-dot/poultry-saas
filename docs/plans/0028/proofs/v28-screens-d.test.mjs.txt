// مراجعة إصلاحات شاشات 0028 (مؤقّت، يُحذف): هجوم على S2 وS4 وS1 والإغلاق بطلب مستعاد. كلّ فحص يقول ما يجب أن يحدث.
import { readFileSync } from 'node:fs';

    const fresh = (who, rpcs = {}) => {

          const count = log.filter(([name]) => name === fn).length;
          return Promise.resolve().then(() => handler(args, count));
        },
      };

    };
    const named = (log, fn) => log.filter(([name]) => name === fn).map(([, args]) => args);

      m.flushSync(() => mounted.unmount());
      body().textContent = '';
      await settle();
    };
    const step = async (label, fn) => {
      session.map.clear();
      local.map.clear();

      }
      globalThis.Date = RealDate;

    };

    const deviceClock = (at, tz) => {
      process.env.TZ = tz;
      const clock = { now: at };
      globalThis.Date = class extends RealDate {
        constructor(...args) {
          if (args.length === 0) super(clock.now);
          else super(...args);
        }

          return clock.now;

    };

    // ── D1 S2 والسباق: أوّل حفظ وصل الخادم وضاع جوابه، ثمّ أكمل المدير التذكرة من جهازه، ثمّ «احذف المسوّدة» هنا ─────────────────────────
    for (const [role, refusal] of [

      ['hall_worker', ['forbidden', 'لا تملك صلاحية هذا الإجراء. اطلب من مالك المنشأة تغيير دورك إن كنت تحتاج إليه.']],

        );
      });

        ['عُدّت القاعة يوم الإغلاق', 'ملاحظة كتبها المدير'],

    await step('D5', async () => {

      await click(byAction('close-retry')[0]);

      await settle();
      await click(byAction('ticket-weigh-empty')[0]);
      await padType('7000');
      await click(within(withAttr('ticket-stop', 'data-stop', stopId), 'ticket-weigh-gross')[0]);

      const stale = textOf('ticket-stale');
      t.same(stale?.includes('من جهاز آخر') ?? null, false, `D6 «تغيّرت» بعد حفظٍ ضاع جوابه من هذا الجهاز نفسه لا تقول «حُفظت من جهاز آخر» (${stale?.slice(0, 100)})`);

    // ── D7 S7: «أعد فتح الدورة» لدورة نُقل علفها ثمّ أُرجع إلى مخزن (return_hall_carry → FR بمصدر النقل): reopen_flock يلغي ذلك المستند أيضاً ─────

        await settle();
        await unmount();
        for (const [label, who, ticket] of [
          ['manager org-1', sessionOf('manager'), done],

          ['manager org-2', sessionOf('manager', { orgId: 'org-2' }), done],

        }
      } catch (error) {
        t.ok(false, `D8: رمى ${error?.stack ?? error}`);
      }
      await unmount();
      t.same(
        seen,
        [
          ['manager org-1', 1, 1],

        'D8 ما كُتب ولم يُحفظ يظهر لكاتبه في منشأته وعلى تذكرته وحدها',
      );
    }
    t.ok(true, 'zz-v28-d ran');
  } finally {
    globalThis.Date = RealDate;
    if (realTz === undefined) delete process.env.TZ;
    else process.env.TZ = realTz;
    if (root) {

        m?.flushSync(() => root.unmount());

        // swallow-ok

    }
    for (const [name, descriptor] of [['sessionStorage', saved.session], ['localStorage', saved.local]]) {

      else delete globalThis[name];
    }
    delete globalThis.__h8;
    restore();
    void all;

  }
